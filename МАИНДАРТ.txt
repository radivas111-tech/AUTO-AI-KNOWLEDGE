const List<String> serverEndpoints = [

String cleanUrl(String brokenUrl) {
  String cleaned = brokenUrl;

  if (cleaned.contains("..")) {
    cleaned = cleaned.replaceAll("..", ".");
  }

  if (cleaned.startsWith("http.//")) {
    cleaned = cleaned.replaceFirst("http.//", "http://");
  }

  if (cleaned.endsWith(".")) {
    cleaned = cleaned.substring(0, cleaned.length - 1);
  }

  return cleaned;
}

Future<http.Response> postWithFallback(
  String route,
  Map<String, dynamic> bodyJson,
) async {
  Object? lastException;

  for (final endpoint in serverEndpoints) {
    try {
      final cleanEndpoint = cleanUrl(endpoint);
      final targetRoute = route.startsWith("/") ? route : "/$route";

      final response = await http
          .post(
            Uri.parse("$cleanEndpoint$targetRoute"),
            headers: {"Content-Type": "application/json"},
            body: jsonEncode(bodyJson),
          )
          .timeout(const Duration(seconds: 7));

      return response;
    } catch (e) {
      lastException = e;
    }
  }

  throw lastException ?? Exception("РћС€РёР±РєР° РїРѕРґРєР»СЋС‡РµРЅРёСЏ Рє СЃРµСЂРІРµСЂСѓ API");
}

  Future<String> _fetchWireGuardConfig() async {
    var token = await _secureStorage.read(
      key: _apiTokenStorageKey,
    );

    if (token == null || token.isEmpty) {
      token = await _requestApiToken();
    }

    if (token == null || token.isEmpty) {
      throw Exception('API token entry cancelled');
    }

    final response = await http.get(
      Uri.parse(_devApiUrl),
      headers: {
        'Authorization': 'Bearer $token',
        'Accept': 'application/json',
      },
    ).timeout(const Duration(seconds: 20));

    if (response.statusCode == 401) {
      await _secureStorage.delete(key: _apiTokenStorageKey);
      throw Exception('API token is invalid or expired');
    }

    if (response.statusCode == 403) {
      throw Exception('DEV client or VPN node is not ready');
    }

    if (response.statusCode != 200) {
      throw Exception('DEV API returned an error');
    }

    final decoded = jsonDecode(response.body);

    if (decoded is! Map<String, dynamic>) {
      throw Exception('Invalid DEV API response');
    }

    final config = decoded['config'];

    if (config is! String ||
        config.trim().isEmpty ||
        !config.contains('[Interface]') ||
        !config.contains('[Peer]')) {
      throw Exception('Invalid WireGuard configuration');
    }

    return config;
  }

  void _startWireGuardStatusPolling() {
    _wireGuardStatusTimer?.cancel();
    var consecutiveDownChecks = 0;

    _wireGuardStatusTimer = Timer.periodic(
      const Duration(seconds: 2),
      (timer) async {
        if (!mounted) {
          timer.cancel();
          _wireGuardStatusTimer = null;
          return;
        }

        if (_wireGuardStatusCheckInProgress) return;
        _wireGuardStatusCheckInProgress = true;

        try {
          final status = await _getWireGuardStatus();
          final state = status['state']?.toString().toUpperCase() ?? 'DOWN';

          if (!mounted) {
            timer.cancel();
            _wireGuardStatusTimer = null;
            return;
          }

          if (state != 'UP') {
            consecutiveDownChecks++;

            if (consecutiveDownChecks >= 5) {
              timer.cancel();
              _wireGuardStatusTimer = null;
              setState(() {
                _isConnected = false;
                _tunnelStatusMessage = 'WireGuard disconnected';
              });
            }
            return;
          }

          consecutiveDownChecks = 0;

          final handshakeValue = status['latestHandshakeEpochMillis'];
          final handshake = int.tryParse(
                handshakeValue?.toString() ?? '0',
              ) ??
              0;

          final connected = handshake > 0;
          final message = connected
              ? 'WireGuard handshake received'
              : 'Tunnel started; waiting for handshake';

          if (_isConnected != connected ||
              _tunnelStatusMessage != message) {
            setState(() {
              _isConnected = connected;
              _tunnelStatusMessage = message;
            });
          }
        } catch (_) {
          // Keep the last known state if a status request temporarily fails.
        } finally {
          _wireGuardStatusCheckInProgress = false;
        }
      },
    );
  }
  Future<Map<String, dynamic>> _connectWireGuard(String config) async {
    final result = await _wireGuardChannel.invokeMapMethod<String, dynamic>(
      'connect',

    return result ?? <String, dynamic>{};
  }
